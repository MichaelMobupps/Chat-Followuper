node -e '
import("node:fs").then(({ default: fs }) => {
  const p = "/tmp/integration-1-7-message.mjs";
  let src = fs.readFileSync(p, "utf8");
  const before = src;

  // 1. Add phone param to createProspect signature
  const sigRe = /async function createProspect\(\{\s*([^}]*?)\s*\}\)/;
  const m = src.match(sigRe);
  if (!m) { console.error("[FAIL] createProspect signature not found"); process.exit(2); }
  if (m[1].includes("phone")) {
    console.log("[SKIP] phone param already in signature");
  } else {
    const params = m[1].trimEnd().replace(/,\s*$/, "");
    src = src.replace(sigRe, `async function createProspect({ ${params}, phone = "+919900000111" })`);
    console.log("[APPLY] signature: added phone param with default");
  }

  // 2. Replace body literal with variable. Heuristic: first occurrence whose preceding 300 chars do NOT contain "createProspect({"
  const litRe = /phone:\s*"\+919900000111"/g;
  const hits = [...src.matchAll(litRe)];
  console.log(`[INFO] phone-literal occurrences: ${hits.length}`);
  let bodyReplaced = false;
  for (const h of hits) {
    const ctx = src.substring(Math.max(0, h.index - 300), h.index);
    if (!ctx.includes("createProspect({")) {
      src = src.substring(0, h.index) + "phone" + src.substring(h.index + h[0].length);
      console.log("[APPLY] body: literal replaced with variable reference");
      bodyReplaced = true;
      break;
    }
  }
  if (!bodyReplaced && hits.length === 0) console.log("[SKIP] body literal not found (already patched?)");

  // 3. Add distinct phones to 2nd and 3rd createProspect call sites
  const callRe = /await createProspect\(\{\s*([^}]*?)\s*\}\)/g;
  const calls = [...src.matchAll(callRe)];
  console.log(`[INFO] createProspect call sites: ${calls.length}`);
  if (calls.length >= 3) {
    const phones = { 1: "+919900000222", 2: "+919900000333" };
    // Process in reverse so earlier indices stay valid
    for (const i of [2, 1]) {
      const c = calls[i];
      if (c[1].includes("phone")) { console.log(`[SKIP] call ${i+1} already has phone`); continue; }
      const args = c[1].trimEnd().replace(/,\s*$/, "");
      const newCall = `await createProspect({ ${args}, phone: "${phones[i]}" })`;
      src = src.substring(0, c.index) + newCall + src.substring(c.index + c[0].length);
      console.log(`[APPLY] call site ${i+1}: phone "${phones[i]}"`);
    }
  } else if (calls.length > 0) {
    console.log("[WARN] expected ≥3 call sites, found " + calls.length + " — review manually");
  }

  if (src === before) console.log("[NOOP] no changes");
  else { fs.writeFileSync(p, src); console.log("[DONE]"); }
});
'
